#define FAN_RELAY      4
#define EXHAUST_RELAY  19
#define LIGHT_RELAY    23
#define HEATER_RELAY   5

float temp = 0;
float hum = 0;
int light = 0;

void setup() {
  Serial.begin(9600);

  pinMode(FAN_RELAY, OUTPUT);
  pinMode(EXHAUST_RELAY, OUTPUT);
  pinMode(LIGHT_RELAY, OUTPUT);
  pinMode(HEATER_RELAY, OUTPUT);

  // Active LOW relay board
  digitalWrite(FAN_RELAY, HIGH);
  digitalWrite(EXHAUST_RELAY, HIGH);
  digitalWrite(LIGHT_RELAY, HIGH);
  digitalWrite(HEATER_RELAY, HIGH);
}

void loop() {

  if (Serial.available()) {
    String data = Serial.readStringUntil('\n');

    int comma1 = data.indexOf(',');
    int comma2 = data.indexOf(',', comma1 + 1);

    if (comma1 > 0 && comma2 > 0) {
      temp = data.substring(0, comma1).toFloat();
      hum = data.substring(comma1 + 1, comma2).toFloat();
      light = data.substring(comma2 + 1).toInt();

      Serial.print("Temp: ");
      Serial.print(temp);
      Serial.print(" Humidity: ");
      Serial.print(hum);
      Serial.print(" Light: ");
      Serial.println(light);
    }
  }

  // Fan ON if very hot
  digitalWrite(FAN_RELAY, temp > 35 ? LOW : HIGH);

  // Exhaust ON if humidity high
  digitalWrite(EXHAUST_RELAY, hum > 65 ? LOW : HIGH);

  // Light ON if dark
  digitalWrite(LIGHT_RELAY, light < 200 ? LOW : HIGH);

  // Heater ON if cold and dry
  digitalWrite(HEATER_RELAY, (temp < 20 && hum < 50) ? LOW : HIGH);

  delay(100);
}



























#include <DHT.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <SoftwareSerial.h>

#define DHTPIN 2
#define DHTTYPE DHT11

DHT dht(DHTPIN, DHTTYPE);
LiquidCrystal_I2C lcd(0x27, 16, 2);

// TX only: UNO pin 8 -> ESP32 GPIO16
SoftwareSerial espSerial(-1, 8);

int ldrPin = A0;

void setup() {
  Serial.begin(9600);      // Dashboard / Python
  espSerial.begin(9600);   // ESP32 relay controller

  dht.begin();

  lcd.init();
  lcd.backlight();

  lcd.setCursor(0, 0);
  lcd.print("AI Microgrid");
  lcd.setCursor(0, 1);
  lcd.print("Controller Boot");
  delay(2000);

  lcd.clear();
}

void loop() {

  float temp = dht.readTemperature();
  float hum = dht.readHumidity();
  int light = analogRead(ldrPin);

  if (isnan(temp) || isnan(hum)) {
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Sensor Error");

    Serial.println("Sensor Error");
    delay(1000);
    return;
  }

  // LCD
  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print("T:");
  lcd.print(temp, 1);
  lcd.print("C H:");
  lcd.print(hum, 0);

  lcd.setCursor(0, 1);
  lcd.print("L:");
  lcd.print(light);

  if (temp > 35) {
    lcd.print(" HOT");
  } else if (light < 200) {
    lcd.print(" DARK");
  }

  // Dashboard output. KEEP THIS EXACT.
  Serial.print("Temp: ");
  Serial.print(temp, 1);
  Serial.print(" C | Humidity: ");
  Serial.print(hum, 0);
  Serial.print(" % | Light: ");
  Serial.println(light);

  // ESP32 output: simple CSV
  espSerial.print(temp, 1);
  espSerial.print(",");
  espSerial.print(hum, 0);
  espSerial.print(",");
  espSerial.println(light);

  delay(1000);
}
